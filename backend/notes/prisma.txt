// ---------- Tenancy & Identity ----------
model Tenant {
  id                    String   @id @default(uuid())
  name                  String
  domain                String?  @unique
  plan_type             String?
  status                String?
  default_admin_user_id String?  @unique
  storage_limit_mb      Int?     @default(0)
  created_at            DateTime @default(now())
  updated_at            DateTime @updatedAt

  users          User[]
  userOrgMap     UserOrgMap[]
  roles          Role[]
  sessions       Session[]
  auditLogs      AuditLog[]
  adminAuditLogs AdminAuditLog[]
  mailAccounts   MailAccount[]
  mailServers    MailServer[]
  syncAudits     SyncAudit[]
  contacts       Contact[]
  imports        Import[]
  emailTemplates EmailTemplate[]
  emailCampaigns EmailCampaign[]
  threadHistory  ThreadHistory[]
  trackingLinks  TrackingLink[]
  usageRecords   UsageRecord[]
  invoices       Invoice[]
  // tenantPolicies         TenantPolicy?
  activities     Activity[]

  adminUser      User?          @relation("TenantAdmin", fields: [default_admin_user_id], references: [id])
  emailMessages  EmailMessage[]
  bounceReports  BounceReport[] // for proper reverse relation
  tenantPolicies TenantPolicy[] // change to one-to-many
}

model User {
  id            String    @id @default(uuid())
  tenant_id     String?
  email         String    @unique
  full_name     String?
  username      String?
  password_hash String?
  auth_provider String?
  role          String?
  status        String?
  last_login_at DateTime?
  created_at    DateTime  @default(now())
  updated_at    DateTime  @updatedAt
  access_token  String?
  refresh_token String?
  ip_address    String?
  user_agent    String?

  tenant          Tenant?         @relation(fields: [tenant_id], references: [id])
  userOrgMap      UserOrgMap[]
  sessions        Session[]
  auditLogs       AuditLog[]
  adminAuditLogs  AdminAuditLog[]
  mailAccounts    MailAccount[]
  syncAudits      SyncAudit[]
  contactsAdded   Contact[]       @relation("AddedBy")
  importsUploaded Import[]        @relation("UploadedBy")
  emailTemplates  EmailTemplate[]
  emailCampaigns  EmailCampaign[]
  tenantPolicies  TenantPolicy[]
  activities      Activity[]

  tenantAdminOf Tenant[] @relation("TenantAdmin")
}

model UserOrgMap {
  id          String   @id @default(uuid())
  user_id     String
  tenant_id   String
  role_in_org String?
  created_at  DateTime @default(now())

  user   User   @relation(fields: [user_id], references: [id])
  tenant Tenant @relation(fields: [tenant_id], references: [id])
}

// ---------- RBAC ----------
model Role {
  id          String   @id @default(uuid())
  tenant_id   String
  name        String?
  description String?
  created_at  DateTime @default(now())

  tenant      Tenant           @relation(fields: [tenant_id], references: [id])
  permissions RolePermission[]
}

model RolePermission {
  id             String   @id @default(uuid())
  role_id        String
  permission_key String?
  created_at     DateTime @default(now())

  role Role @relation(fields: [role_id], references: [id])
}

// ---------- Sessions & Audits ----------
model Session {
  id          String    @id @default(uuid())
  user_id     String
  tenant_id   String
  ip_address  String?
  device_info String?
  created_at  DateTime  @default(now())
  expires_at  DateTime?
  is_active   Boolean   @default(true)

  user   User         @relation(fields: [user_id], references: [id])
  tenant Tenant       @relation(fields: [tenant_id], references: [id])
  logs   SessionLog[]
}

model SessionLog {
  id         String   @id @default(uuid())
  session_id String
  event      String?
  provider   String?
  details    Json?
  created_at DateTime @default(now())

  session Session @relation(fields: [session_id], references: [id])
}

  tenant Tenant? @relation(fields: [tenant_id], references: [id])
}
